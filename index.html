async function submitDeposit(){ const amount = Number(document.getElementById("depAmount").value); const ref = document.getElementById("depRef").value.trim(); const operator = PAYMENTS[ Number(document.getElementById("depOperator").value) ]; if (!user) { alert("Veuillez vous connecter."); return; } if (amount < 15000) { alert("Le dépôt minimum est de 15 000 FC."); return; } try { // Recherche du moyen de paiement dans Supabase const { data: paymentMethod, error: paymentError } = await sb .from("kin_payment_methods") .select("id") .eq("number", operator.number) .maybeSingle(); if (paymentError) throw paymentError; if (!paymentMethod) { throw new Error( "Le moyen de paiement sélectionné n'est pas enregistré." ); } // Création de la demande de dépôt const { error } = await sb .from("kin_deposit_requests") .insert({ user_id: user.id, amount: amount, payment_method_id: paymentMethod.id, transaction_id: ref || null, status: "pending" }); if (error) throw error; closeModal(); alert( "✅ Demande de dépôt envoyée avec succès.\n\n" + "Montant : " + amount.toLocaleString("fr-FR") + " FC\n\n" + "Votre demande est maintenant en attente de validation par l'administration." ); } catch (e) { console.error("DEPOT:", e); alert( "❌ Impossible d'envoyer le dépôt.\n\n" + e.message ); } }

async function adminAction(type, id, value) { if (!isAdmin) { alert("Accès administrateur refusé."); return; } const labels = { deposit: value === "approved" ? "valider ce dépôt" : "rejeter ce dépôt", withdrawal: value === "approved" ? "confirmer ce retrait" : "rejeter ce retrait", member: value ? "bloquer ce membre" : "débloquer ce membre" }; if (!confirm("Confirmez-vous : " + labels[type] + " ?")) return; try { let result; if (type === "deposit") { result = await sb.rpc("admin_set_deposit_status", { p_request_id: id, p_status: value, p_note: null }); } else if (type === "withdrawal") { result = await sb.rpc("admin_set_withdrawal_status", { p_request_id: id, p_status: value, p_note: null }); } else { result = await sb.rpc("admin_set_member_ban", { p_user_id: id, p_banned: value }); } if (result.error) throw result.error; alert("Opération enregistrée."); await loadAdminData(); } catch (error) { alert("Échec : " + (error.message || "erreur inconnue") + "\nVérifiez les fonctions SQL et les autorisations."); } } async function renderAdmin() { if (!isAdmin) { alert("Cette page est réservée à l'administration."); page("profile"); return; } document.getElementById("content").innerHTML = `

Résumé

Chargement…

Membres inscrits

Chargement…

Dépôts en attente

Chargement…

Retraits en attente

Chargement…

`; await loadAdminData(); } async function loadAdminData() { if (!isAdmin) return; const members = document.getElementById("members"); const deposits = document.getElementById("deposits"); const withdrawals = document.getElementById("withdrawals"); const stats = document.getElementById("adminStats"); if (!members || !deposits || !withdrawals || !stats) return; members.innerHTML = "Chargement…"; deposits.innerHTML = "Chargement…"; withdrawals.innerHTML = "Chargement…"; const [m, d, w] = await Promise.all([ sb.from("profiles").select("*").order("created_at", { ascending: false }), sb.from("kin_deposit_requests").select("*") .order("created_at", { ascending: false }), sb.from("kin_withdrawal_requests").select("*") .order("created_at", { ascending: false }) ]); if (m.error) members.innerHTML = msg(m.error.message, "err"); else { const list = m.data || []; stats.innerHTML = `

Dépôts en attente${(d.data || []).filter(x => x.status === "pending").length}
Retraits en attente${(w.data || []).filter(x => x.status === "pending").length}
`; members.innerHTML = list.map(x => `
${esc(x.full_name || "Membre")}
${esc(x.id)}
Email : ${esc(x.email || "Non disponible")}

Statut : ${x.is_banned ? "Bloqué" : "Actif"}
`).join("") || "Aucun membre trouvé."; } if (d.error) deposits.innerHTML = msg(d.error.message, "err"); else { deposits.innerHTML = (d.data || []) .filter(x => x.status === "pending") .map(x => `
${money(x.amount)}
Membre : ${esc(x.user_id)}
Référence : ${esc(x.transaction_id || x.reference || "—")}
Statut : En attente
`).join("") || "Aucun dépôt en attente."; } if (w.error) withdrawals.innerHTML = msg(w.error.message, "err"); else { withdrawals.innerHTML = (w.data || []) .filter(x => x.status === "pending") .map(x => `
${money(x.amount)}
Membre : ${esc(x.user_id)}

`).join("") || "Aucun retrait en attente."; } }

async function initializeAccount(user){ currentUser = user; isAdmin = false; profile = null; const email = (user.email || "").toLowerCase().trim(); // Vérification du compte administrateur dans Supabase. const adminResult = await sb .from("kb_admins") .select("user_id") .eq("user_id", user.id) .maybeSingle(); if (adminResult.error) { console.error("Erreur de vérification administrateur:", adminResult.error); throw new Error( "Impossible de vérifier les droits administrateur. Vérifiez les autorisations de la table kb_admins dans Supabase." ); } // Le compte doit être enregistré dans kb_admins ET utiliser cet e-mail. isAdmin = email === "angenabukuze@gmail.com" && !!adminResult.data && adminResult.data.user_id === user.id; // Charger le profil avec les vrais noms de colonnes. const profileResult = await sb .from("kb_profiles") .select("user_id,full_name,phone,balance_fc,created_at") .eq("user_id", user.id) .maybeSingle(); if (profileResult.error) { throw profileResult.error; } if (!profileResult.data) { const metadata = user.user_metadata || {}; const inserted = await sb .from("kb_profiles") .insert({ user_id: user.id, full_name: metadata.full_name || "", phone: metadata.phone || "", balance_fc: 0 }) .select("user_id,full_name,phone,balance_fc,created_at") .single(); if (inserted.error) { throw inserted.error; } profile = inserted.data; } else { profile = profileResult.data; } // Afficher les informations du compte. document.getElementById("connectionStatus").textContent = isAdmin ? "ADMINISTRATEUR" : "CONNECTÉ"; document.getElementById("profileRole").textContent = isAdmin ? "ADMINISTRATEUR" : "MEMBRE"; document.getElementById("profileName").textContent = profile.full_name || "Membre"; document.getElementById("profilePhone").textContent = profile.phone || "—"; document.getElementById("profileEmail").textContent = user.email || "—"; document.getElementById("homeUserName").textContent = profile.full_name || "Membre"; document.getElementById("homeBalance").textContent = Number(profile.balance_fc || 0).toLocaleString("fr-FR") + " FC"; // Afficher ou masquer le panneau administrateur. document.getElementById("adminPanel").classList.toggle( "hidden", !isAdmin ); // Ouvrir l'application. document.getElementById("authPage").classList.add("hidden"); document.getElementById("app").classList.remove("hidden"); document.getElementById("bottomNav").classList.remove("hidden"); window.go("accueil"); // Charger immédiatement les demandes si l'utilisateur est administrateur. if (isAdmin) { await window.loadAdmin(); } }

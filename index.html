window.loadHistory = async function () { const el = document.getElementById("historyList"); if (!currentUser) { el.innerHTML = "

Connectez-vous pour voir votre historique.

"; return; } el.innerHTML = "

Chargement de l'historique...

"; try { const uid = currentUser.id; const [transactions, deposits, withdrawals] = await Promise.all([ sb.from("kb_transactions") .select("id,transaction_type,amount_fc,description,created_at") .eq("user_id", uid) .order("created_at", { ascending: false }) .limit(100), sb.from("kb_deposits") .select("id,amount_fc,payment_method,reference,status,created_at") .eq("user_id", uid) .order("created_at", { ascending: false }) .limit(100), sb.from("kb_withdrawals") .select("id,amount_fc,payment_method,destination_phone,status,created_at") .eq("user_id", uid) .order("created_at", { ascending: false }) .limit(100) ]); if (transactions.error) throw transactions.error; if (deposits.error) throw deposits.error; if (withdrawals.error) throw withdrawals.error; const rows = []; (transactions.data || []).forEach(t => { rows.push({ id: "T-" + t.id, date: t.created_at, type: t.transaction_type || "Transaction", amount: Number(t.amount_fc || 0), details: t.description || "Mouvement de compte", status: "Enregistrée" }); }); (deposits.data || []).forEach(d => { rows.push({ id: "D-" + d.id, date: d.created_at, type: "Dépôt", amount: Number(d.amount_fc || 0), details: (d.payment_method || "") + (d.reference ? " · Référence : " + d.reference : ""), status: d.status || "pending" }); }); (withdrawals.data || []).forEach(w => { rows.push({ id: "R-" + w.id, date: w.created_at, type: "Retrait", amount: Number(w.amount_fc || 0), details: (w.payment_method || "") + (w.destination_phone ? " · " + w.destination_phone : ""), status: w.status || "pending" }); }); rows.sort((a, b) => new Date(b.date || 0).getTime() - new Date(a.date || 0).getTime() ); if (!rows.length) { el.innerHTML = "

Aucun mouvement enregistré pour ce compte.

"; return; } const statusLabel = status => ({ pending: "En attente", approved: "Approuvé", rejected: "Refusé", completed: "Terminé" }[String(status).toLowerCase()] || status); el.innerHTML = rows.map(t => `
${esc(t.type)} — ${fmt(t.amount)}
${esc(t.details)}
Statut : ${esc(statusLabel(t.status))}
${t.date ? esc(new Date(t.date).toLocaleString("fr-FR")) : ""}
`).join(""); } catch (e) { console.error("Erreur historique :", e); el.innerHTML = "

Impossible de charger l'historique. Vérifiez les autorisations Supabase.

"; } };

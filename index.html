async function loadAdmin() { if (!user || (user.email || "").toLowerCase() !== ADMIN_EMAIL) { msg("Accès administrateur refusé.", "error"); return; } const depositBox = document.getElementById("adminDeposits"); const withdrawalBox = document.getElementById("adminWithdrawals"); depositBox.textContent = "Chargement des dépôts..."; withdrawalBox.textContent = "Chargement des retraits..."; const [deposits, withdrawals] = await Promise.all([ sb.from("kb_deposits") .select("*") .eq("status", "pending") .order("created_at", { ascending: true }) .limit(100), sb.from("kb_withdrawals") .select("*") .eq("status", "pending") .order("created_at", { ascending: true }) .limit(100) ]); depositBox.replaceChildren(); withdrawalBox.replaceChildren(); if (deposits.error) { depositBox.textContent = "Erreur : " + deposits.error.message; } else if (!deposits.data.length) { depositBox.textContent = "Aucun dépôt en attente."; } else { deposits.data.forEach(item => { const card = document.createElement("div"); card.className = "offer"; const details = document.createElement("p"); details.textContent = "Montant : " + money(item.amount) + " | Moyen : " + (item.payment_method || "") + " | Référence : " + (item.payment_reference || "") + " | Statut : " + item.status; const approve = document.createElement("button"); approve.textContent = "APPROUVER LE DÉPÔT"; approve.onclick = () => reviewDeposit(item.id, true); const reject = document.createElement("button"); reject.className = "red"; reject.textContent = "REFUSER LE DÉPÔT"; reject.onclick = () => reviewDeposit(item.id, false); card.append(details, approve, reject); depositBox.appendChild(card); }); } if (withdrawals.error) { withdrawalBox.textContent = "Erreur : " + withdrawals.error.message; } else if (!withdrawals.data.length) { withdrawalBox.textContent = "Aucun retrait en attente."; } else { withdrawals.data.forEach(item => { const card = document.createElement("div"); card.className = "offer"; const details = document.createElement("p"); details.textContent = "Montant : " + money(item.amount) + " | Moyen : " + (item.payment_method || "") + " | Numéro : " + (item.phone_number || "") + " | Statut : " + item.status; const approve = document.createElement("button"); approve.textContent = "APPROUVER LE RETRAIT"; approve.onclick = () => reviewWithdrawal(item.id, true); const reject = document.createElement("button"); reject.className = "red"; reject.textContent = "REFUSER LE RETRAIT"; reject.onclick = () => reviewWithdrawal(item.id, false); card.append(details, approve, reject); withdrawalBox.appendChild(card); }); } }
